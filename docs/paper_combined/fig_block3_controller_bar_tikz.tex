\usepackage{lmodern}
\usepackage{amsmath}

\pgfplotstableread[col sep=comma]{../../reports/block3_transfer_matrix.csv}\dtransfer

\begin{document}
\begin{tikzpicture}
\begin{axis}[
    paperaxis, paperbars, bottomlegend,
    width=10.5cm, height=6.8cm,
    title={Controller-side transfer vs the pre-specified threshold},
    ylabel={$m_s$ (lower better)},
    ymin=0, ymax=1.15,
    xtick={0,1,2},
    xticklabels={heat pump, hydronic, commercial},

    bar width=13pt,

  \legend{$m_s^{\mathrm{RL}}$ (frozen hybrid), $m_s^{\mathrm{PI}}$ (baseline)}

  % PASS/FAIL verdict printed straight from the CSV column (data-driven)
  \addplot[draw=none, forget plot, mark=none, nodes near coords,
           point meta=explicit symbolic,
           nodes near coords style={anchor=south, yshift=3pt, xshift=-13pt,
             font=\footnotesize\bfseries, text=cInk}]
    table[x expr=\coordindex, y=m_s_rl, meta=none_controller_verdict]{\dtransfer};
  \node[font=\scriptsize, text=cAccurate, anchor=east] at (axis cs:2.45,0.80)
    {$\tau_k=1.25\,m_s^{\mathrm{PI}}$};
