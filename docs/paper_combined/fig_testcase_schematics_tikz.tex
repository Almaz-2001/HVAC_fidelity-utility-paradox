% #1 xshift #2 yshift #3 title #4 subtitle #5 source #6 distribution #7 action
\newcommand{\tcase}[7]{%
\begin{scope}[shift={(#1,#2)}]
  \node[font=\sffamily\bfseries] at (40mm,3mm) {#3};
  \node[font=\sffamily\footnotesize, text=cMut] at (40mm,-1.5mm) {#4};
  \node[zone, anchor=north west] at (24mm,-8mm) (Z) {\textbf{Zone}\\[0.4mm]%
     {\footnotesize $C_{\mathrm{zon}}\dot T=\dot Q_{\mathrm{hyd}}+(T_{\mathrm{amb}}{-}T)/R$}\\[0.4mm]%
     {\scriptsize\itshape sensors: $T_{\mathrm{zon}}$, CO$_2$, power}};
  \node[src,  anchor=north west] at (0mm,-36mm)  (S) {\footnotesize #5};
  \node[dist, anchor=north west] at (36mm,-36mm) (D) {\footnotesize #6};
  \draw[farr] (S.east) -- (D.west);
  \draw[farr] (D.north) -- node[right=0.3mm,font=\scriptsize,text=cSrc]{$\dot Q_{\mathrm{hyd}}$} (D.north|-Z.south);
  \node[font=\scriptsize, text=cMut, anchor=east] at (3mm,-15mm) (dT) {$T_{\mathrm{amb}}$, solar};
  \node[font=\scriptsize, text=cMut, anchor=east] at (3mm,-21mm) (dR) {$R_{\mathrm{env}}$};
  \draw[garr] (dT.east) -- (dT.east-|Z.west);
  \draw[garr] (dR.east) -- ([yshift=-6mm]dR.east-|Z.west);
  \node[font=\sffamily\bfseries\footnotesize, text=cAct, align=center] at (74mm,-11mm) (CA) {controller\\action};
  \node[font=\scriptsize, text=cAct, align=center, text width=26mm] at (74mm,-19mm) (AL) {#7};
  \draw[aarr] (AL.south) to[bend left=18] (D.east);
\end{scope}}
