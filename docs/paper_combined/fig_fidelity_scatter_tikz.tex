\documentclass[border=4pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{amsmath}
\input{_pgfstyle}

\pgfplotstableread[col sep=comma]{../../reports/block2_fidelity_utility_scatter.csv}\dsc
\newcommand{\getc}[3]{\pgfplotstablegetelem{#2}{#3}\of\dsc \pgfmathsetmacro{#1}{\pgfplotsretval}}
% peak (filled) + typical (open) markers with thin y error bars, for one model row
\newcommand{\modelpts}[2]{% #1=row #2=colour
  \getc{\xr}{#1}{rmse_24h_c}%
  \getc{\yp}{#1}{m_s_peak}\getc{\sp}{#1}{std_peak}%
  \getc{\yt}{#1}{m_s_typ}\getc{\st}{#1}{std_typ}%
  \addplot[#2, only marks, mark=*, mark size=3pt, thick, forget plot,
           error bars/.cd, y dir=both, y explicit, error bar style={line width=0.5pt}]
    coordinates {(\xr,\yp) +- (0,\sp)};
  \addplot[#2, only marks, mark=o, mark size=3pt, thick, forget plot,
           error bars/.cd, y dir=both, y explicit, error bar style={line width=0.5pt}]
    coordinates {(\xr,\yt) +- (0,\st)};
}
% shared per-panel decoration: usable band, collapse band, threshold lines
\newcommand{\zones}{%
  \fill[cV3, fill opacity=0.14] (axis cs:0.5,-0.05) rectangle (axis cs:1.75,\MSusable);
  \fill[cAccurate, fill opacity=0.06] (axis cs:0.5,\MScollapse) rectangle (axis cs:1.75,1.4);
  \addplot[cAccurate, dashed, line width=1pt, domain=0.5:1.75, samples=2, forget plot] {\MScollapse};
  \addplot[cV3, dotted, line width=1pt, domain=0.5:1.75, samples=2, forget plot] {\MSusable};
  \node[font=\footnotesize, text=cAccurate, anchor=east, fill=white, fill opacity=0.7, inner sep=1pt] at (axis cs:1.73,1.18) {$\ms=1$ collapse};
  \node[font=\scriptsize, text=cV3!55!black, anchor=west] at (axis cs:0.54,0.045) {usable $\ms<0.1$};
}
