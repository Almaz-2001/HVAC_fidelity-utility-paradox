\newcommand{\getc}[3]{\pgfplotstablegetelem{#2}{#3}\of\ds \pgfmathsetmacro{#1}{\pgfplotsretval}}
% peak (filled) + typical (open) with y error bars, at controller index #1, colour #2
\newcommand{\sbms}[2]{%
  \pgfmathsetmacro{\xl}{#1-0.15}\pgfmathsetmacro{\xr}{#1+0.15}%
  \getc{\yp}{#1}{ms_peak}\getc{\sp}{#1}{std_peak}\getc{\yt}{#1}{ms_typ}\getc{\st}{#1}{std_typ}%
  \addplot[#2,only marks,mark=*,mark size=3pt,thick,error bars/.cd,y dir=both,y explicit,error bar style={line width=0.5pt}] coordinates {(\xl,\yp) +- (0,\sp)};
  \addplot[#2,only marks,mark=o,mark size=3pt,thick,error bars/.cd,y dir=both,y explicit,error bar style={line width=0.5pt}] coordinates {(\xr,\yt) +- (0,\st)};
}
\newcommand{\sbviol}[2]{%
  \pgfmathsetmacro{\xl}{#1-0.15}\pgfmathsetmacro{\xr}{#1+0.15}%
  \getc{\vp}{#1}{viol_peak}\getc{\vt}{#1}{viol_typ}%
  \addplot[#2,only marks,mark=*,mark size=3pt] coordinates {(\xl,\vp)};
  \addplot[#2,only marks,mark=o,mark size=3pt] coordinates {(\xr,\vt)};
}
